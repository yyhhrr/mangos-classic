cmake_minimum_required(VERSION 3.15)
project(gnomeregan_escort_tests LANGUAGES CXX)

if(NOT EXISTS "${SCRIPT_UNDER_TEST}")
    message(FATAL_ERROR "Pass the actual gnomeregan.cpp as SCRIPT_UNDER_TEST")
endif()

# Compile the real quest AI definitions against small engine doubles. There is
# no separately maintained copy of the escort logic in these tests.
file(READ "${SCRIPT_UNDER_TEST}" script_source)
string(REPLACE "\r\n" "\n" script_source "${script_source}")
string(FIND "${script_source}" "\nenum\n{\n    QUEST_A_FINE_MESS" begin_offset)
string(FIND "${script_source}" "\nUnitAI* GetAI_npc_kernobee" end_offset)
if(begin_offset LESS 0 OR end_offset LESS begin_offset)
    message(FATAL_ERROR "Cannot locate quest 2904 AI definitions")
endif()
math(EXPR body_length "${end_offset} - ${begin_offset}")
string(SUBSTRING "${script_source}" ${begin_offset} ${body_length} escort_source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/escort_under_test.h" "${escort_source}")

# Keep group failure tests tied to the real engine helpers as well.
get_filename_component(repo_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
function(extract_function source signature output)
    file(READ "${repo_root}/${source}" body)
    string(REPLACE "\r\n" "\n" body "${body}")
    string(FIND "${body}" "${signature}" start)
    if(start LESS 0)
        message(FATAL_ERROR "Missing engine function ${signature}")
    endif()
    string(SUBSTRING "${body}" ${start} -1 tail)
    string(FIND "${tail}" "\n}\n" finish)
    math(EXPR length "${finish} + 3")
    string(SUBSTRING "${tail}" 0 ${length} function_body)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${function_body}\n")
endfunction()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/failure_under_test.h" "")
extract_function("src/game/Entities/Player.cpp" "void Player::FailQuestForGroup(" failure_under_test.h)
extract_function("src/game/AI/ScriptDevAI/base/follower_ai.cpp" "void FollowerAI::JustDied(" failure_under_test.h)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/relations_under_test.h" "")
foreach(name CanAttackSpell CanAttackInCombat CanAttackServerside)
    extract_function("src/game/Entities/Relations.cpp" "bool Unit::${name}(" relations_under_test.h)
endforeach()
file(READ "${repo_root}/src/game/Spells/Spell.h" spell_header)
string(REPLACE "\r\n" "\n" spell_header "${spell_header}")
string(FIND "${spell_header}" "template<class T> inline void Visit(GridRefManager<T>& m)" visitor_start)
string(SUBSTRING "${spell_header}" ${visitor_start} -1 visitor_tail)
string(FIND "${visitor_tail}" "\n#ifdef _MSC_VER" visitor_length)
string(SUBSTRING "${visitor_tail}" 0 ${visitor_length} visitor_body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/aoe_visitor_under_test.h" "${visitor_body}")

add_executable(gnomeregan_escort_tests escort_tests.cpp)
target_compile_features(gnomeregan_escort_tests PRIVATE cxx_std_17)
target_include_directories(gnomeregan_escort_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME quest_2904_alarm_linkage COMMAND gnomeregan_escort_tests)
add_executable(gnomeregan_spell_targets target_tests.cpp)
target_compile_features(gnomeregan_spell_targets PRIVATE cxx_std_17)
target_include_directories(gnomeregan_spell_targets PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME spell_9874_engine_target_eligibility COMMAND gnomeregan_spell_targets)
